## ⚠️ Important Update (2025-10-04)

**Story 1.8 introduces Review Workflow** - Translation workers must now set status to `"in_review"` instead of `"completed"`:

**Changes Required**:
1. Translation worker sets `status: "in_review"` when translation completes (not `"completed"`)
2. Translation worker populates `language_status` JSONB with per-language tracking
3. Each language in `language_status` has: `{status, progress, reviewed, reviewer_id, reviewed_at, review_notes}`
4. Human reviewer approves/rejects via `/api/translations/review/{job_id}/{language_code}` endpoint
5. Only after ALL languages approved → status changes to `"completed"`

See **Story 1.8** for full review workflow specification.

```python
from google.cloud import translate_v2 as translate
import os

class TranslationService:
    def __init__(self):
        """Initialize Google Cloud Translation client"""
        self.client = translate.Client()

    def translate_text(self, text: str, target_language: str, source_language: str = "en") -> dict:
        """
        Translate text using Vertex AI Translation

        Args:
            text: Text to translate
            target_language: ISO 639-1 language code (e.g., 'hi' for Hindi)
            source_language: Source language code (default: 'en')

        Returns:
            {
                "translated_text": "...",
                "source_language": "en",
                "target_language": "hi",
                "character_count": 1234
            }
        """
        if not text or not text.strip():
            raise ValueError("Text cannot be empty")

        # Validate target language
        supported_languages = ['hi', 'bn', 'ta', 'te', 'kn', 'ml']
        if target_language not in supported_languages:
            raise ValueError(f"Unsupported language: {target_language}")

        try:
            # Call Google Cloud Translation API
            result = self.client.translate(
                text,
                target_language=target_language,
                source_language=source_language,
                format_='text'
            )

            return {
                "translated_text": result['translatedText'],
                "source_language": result.get('detectedSourceLanguage', source_language),
                "target_language": target_language,
                "character_count": len(text)
            }

        except Exception as e:
            raise Exception(f"Translation failed: {str(e)}")

    def get_supported_languages(self) -> list:
        """Return list of supported languages"""
        return [
            {"code": "hi", "name": "Hindi"},
            {"code": "bn", "name": "Bengali"},
            {"code": "ta", "name": "Tamil"},
            {"code": "te", "name": "Telugu"},
            {"code": "kn", "name": "Kannada"},
            {"code": "ml", "name": "Malayalam"}
        ]
```

### Step 2: Create Translation API Router

**File**: `backend/api/routers/translations.py` (NEW)

```python
from fastapi import APIRouter, HTTPException
from pydantic import BaseModel
from typing import List, Optional
import uuid
import json
import psycopg2
import os
from supabase import create_client

router = APIRouter(prefix="/api/translations", tags=["translations"])

class TranslationRequest(BaseModel):
    target_languages: List[str]  # ['hi', 'bn', 'ta']
    source_language: str = "en"
    priority: int = 5  # 1-10, higher = more urgent

class TranslationResponse(BaseModel):
    job_id: str
    document_id: str
    status: str
    target_languages: List[str]
    message: str

@router.post("/start/{document_id}", response_model=TranslationResponse)
async def start_translation(document_id: str, request: TranslationRequest):
    """
    Start translation job for document

    Prerequisites:
    - Document must exist
    - Text extraction must be completed
    - Target languages must be valid
    """

    # Validate document exists and has extracted text
    supabase = create_client(os.getenv("SUPABASE_URL"), os.getenv("SUPABASE_SERVICE_KEY"))

    doc_result = supabase.table("documents").select("*").eq("id", document_id).execute()
    if not doc_result.data:
        raise HTTPException(status_code=404, detail="Document not found")

    # Check if text extraction is complete
    extraction_result = supabase.table("text_extraction_results")\
        .select("*")\
        .eq("document_id", document_id)\
        .eq("status", "completed")\
        .execute()

    if not extraction_result.data:
        raise HTTPException(
            status_code=400,
            detail="Text extraction must be completed before translation"
        )

    extracted_text = extraction_result.data[0].get("extracted_text")
    if not extracted_text:
        raise HTTPException(status_code=400, detail="No extracted text found")

    # Validate target languages
    supported_languages = ['hi', 'bn', 'ta', 'te', 'kn', 'ml']
    invalid_languages = [lang for lang in request.target_languages if lang not in supported_languages]
    if invalid_languages:
        raise HTTPException(
            status_code=400,
            detail=f"Unsupported languages: {', '.join(invalid_languages)}"
        )

    # Create translation job
    job_id = str(uuid.uuid4())

    job_data = {
        "id": job_id,
        "document_id": document_id,
        "status": "queued",
        "progress": 0,
        "target_languages": request.target_languages,
        "source_language": request.source_language,
        "character_count": len(extracted_text),
        "estimated_cost": calculate_translation_cost(len(extracted_text), len(request.target_languages))
    }

    supabase.table("translation_results").insert(job_data).execute()

    # Queue translation job via pgmq
    conn = psycopg2.connect(os.getenv("DATABASE_URL"))
    cursor = conn.cursor()

    queue_message = {
        "job_id": job_id,
        "document_id": document_id,
        "target_languages": request.target_languages,
        "source_language": request.source_language,
        "priority": request.priority
    }

    cursor.execute(
        "SELECT pgmq.send(%s, %s::jsonb, %s);",
        ("translation_queue", json.dumps(queue_message), request.priority)
    )

    conn.commit()
    cursor.close()
    conn.close()

    return TranslationResponse(
        job_id=job_id,
        document_id=document_id,
        status="queued",
        target_languages=request.target_languages,
        message=f"Translation job queued for {len(request.target_languages)} languages"
    )


@router.get("/status/{job_id}")
async def get_translation_status(job_id: str):
    """Get translation job status"""
    supabase = create_client(os.getenv("SUPABASE_URL"), os.getenv("SUPABASE_SERVICE_KEY"))

    result = supabase.table("translation_results").select("*").eq("id", job_id).execute()

    if not result.data:
        raise HTTPException(status_code=404, detail="Translation job not found")

    job = result.data[0]

    return {
        "job_id": job_id,
        "status": job["status"],
        "progress": job["progress"],
        "target_languages": job["target_languages"],
        "completed_languages": job.get("completed_languages", []),
        "character_count": job["character_count"],
        "estimated_cost": job.get("estimated_cost", 0)
    }


@router.get("/supported-languages")
async def get_supported_languages():
    """Get list of supported translation languages"""
    return {
        "languages": [
            {"code": "hi", "name": "Hindi"},
            {"code": "bn", "name": "Bengali"},
            {"code": "ta", "name": "Tamil"},
            {"code": "te", "name": "Telugu"},
            {"code": "kn", "name": "Kannada"},
            {"code": "ml", "name": "Malayalam"}
        ]
    }


def calculate_translation_cost(character_count: int, language_count: int) -> float:
    """
    Calculate estimated translation cost
    Google Cloud Translation pricing: $20 per 1M characters
    """
    total_characters = character_count * language_count
    cost_per_million = 20.0
    return (total_characters / 1_000_000) * cost_per_million
```

### Step 3: Create Translation Worker

**File**: `backend/workers/translation_worker.py` (NEW)

```python
import asyncio
import psycopg2
import json
import os
from supabase import create_client
from backend.api.services.translation_service import TranslationService

async def run_translation_worker(worker_id: int):
    """
    Translation worker - processes jobs from translation_queue
    Pattern copied from text_extraction_worker.py
    """
    print(f"[Worker-{worker_id}] Translation worker starting...")

    # Initialize connections
    db_conn = psycopg2.connect(os.getenv("DATABASE_URL"))
    supabase = create_client(os.getenv("SUPABASE_URL"), os.getenv("SUPABASE_SERVICE_KEY"))
    translation_service = TranslationService()

    while True:
        try:
            cursor = db_conn.cursor()

            # Poll queue (visibility timeout: 300 seconds)
            cursor.execute(
                "SELECT * FROM pgmq.read(%s, %s, %s);",
                ("translation_queue", 300, 1)
            )

            result = cursor.fetchone()

            if not result:
                # No jobs in queue, wait 2 seconds
                await asyncio.sleep(2)
                continue

            # Extract message data
            msg_id = result[0]
            message = result[4]  # JSON message body

            job_id = message["job_id"]
            document_id = message["document_id"]
            target_languages = message["target_languages"]
            source_language = message.get("source_language", "en")

            print(f"[Worker-{worker_id}] Processing translation job {job_id}")

            # Update status to processing
            supabase.table("translation_results").update({
                "status": "processing",
                "progress": 0
            }).eq("id", job_id).execute()

            # Get extracted text
            extraction_result = supabase.table("text_extraction_results")\
                .select("extracted_text")\
                .eq("document_id", document_id)\
                .eq("status", "completed")\
                .execute()

            if not extraction_result.data:
                raise Exception("No extracted text found")

            extracted_text = extraction_result.data[0]["extracted_text"]

            # Translate to each target language
            translations = {}
            completed_languages = []

            for i, target_lang in enumerate(target_languages):
                print(f"[Worker-{worker_id}] Translating to {target_lang}...")

                # Translate text
                translation_result = translation_service.translate_text(
                    text=extracted_text,
                    target_language=target_lang,
                    source_language=source_language
                )

                translations[target_lang] = translation_result["translated_text"]
                completed_languages.append(target_lang)

                # Update progress
                progress = int((i + 1) / len(target_languages) * 100)
                supabase.table("translation_results").update({
                    "progress": progress,
                    "completed_languages": completed_languages
                }).eq("id", job_id).execute()

            # Save translations to database
            supabase.table("translation_results").update({
                "status": "completed",
                "progress": 100,
                "translations": translations,
                "completed_languages": completed_languages
            }).eq("id", job_id).execute()

            print(f"[Worker-{worker_id}] Translation job {job_id} completed")

            # Archive message from queue
            cursor.execute(
                "SELECT pgmq.archive(%s, %s);",
                ("translation_queue", msg_id)
            )
            db_conn.commit()

        except Exception as e:
            print(f"[Worker-{worker_id}] Error: {str(e)}")

            # Mark job as failed
            try:
                supabase.table("translation_results").update({
                    "status": "failed",
                    "error_message": str(e)
                }).eq("id", job_id).execute()
            except:
                pass

            # Delete message from queue
            try:
                cursor.execute(
                    "SELECT pgmq.archive(%s, %s);",
                    ("translation_queue", msg_id)
                )
                db_conn.commit()
            except:
                pass

            await asyncio.sleep(2)

        finally:
            if cursor:
                cursor.close()


async def main():
    """Run translation worker pool (4 concurrent workers)"""
    worker_count = int(os.getenv("WORKER_POOL_SIZE", "4"))

    print(f"Starting translation worker pool with {worker_count} workers...")

    workers = [
        asyncio.create_task(run_translation_worker(i))
        for i in range(worker_count)
    ]

    try:
        await asyncio.gather(*workers)
    except KeyboardInterrupt:
        print("\nShutting down translation workers...")


if __name__ == "__main__":
    asyncio.run(main())
```

```sql
-- Translation results table (should already exist)
CREATE TABLE IF NOT EXISTS translation_results (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    document_id UUID NOT NULL REFERENCES documents(id) ON DELETE CASCADE,
    status VARCHAR(50) NOT NULL DEFAULT 'queued',
    progress INTEGER DEFAULT 0,
    target_languages TEXT[] NOT NULL,
    source_language VARCHAR(10) DEFAULT 'en',
    completed_languages TEXT[] DEFAULT '{}',
    translations JSONB,
    character_count INTEGER,
    estimated_cost DECIMAL(10, 4),
    error_message TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Index for faster queries
CREATE INDEX IF NOT EXISTS idx_translation_results_document_id ON translation_results(document_id);
CREATE INDEX IF NOT EXISTS idx_translation_results_status ON translation_results(status);
```
