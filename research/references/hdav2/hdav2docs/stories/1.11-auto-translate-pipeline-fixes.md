## User Story

**As a** User uploading documents with auto-translate enabled
**I want** translation jobs to be automatically created and processed when text extraction completes
**So that** I don't have to manually initiate translation for each target language

---

## Context

**Problem Identified** (from E2E testing 2025-10-05):

During comprehensive E2E testing, discovered that the auto-translate feature is completely broken:
- ✅ Frontend correctly sends `auto_translate: true` with target languages
- ✅ Backend API correctly stores the flag in job metadata
- ❌ Text extraction workers **ignore** the auto_translate flag entirely
- ❌ Translation jobs are **never created** automatically
- ❌ Workflow stops at "in-review" status instead of proceeding to translation pipeline

**Expected Workflow**:
```
Upload (auto_translate: true, languages: [hi, mr, bn])
  ↓
Text Extraction: Queued → Processing → Completed
  ↓
[AUTO-TRIGGER]
  ↓
Translation Jobs Created: 3 jobs (Hindi, Marathi, Bengali)
  ↓
Translation: Queued → Processing → In-Review → Completed
```

**Current Broken Workflow**:
```
Upload (auto_translate: true, languages: [hi, mr, bn])
  ↓
Text Extraction: Queued → Processing → In-Review
  ↓
[STOPS HERE - No translation jobs created] ❌
```

**Additional Critical Issues from E2E Testing**:
1. ✅ Backend translation API uses wrong table name (`translations` instead of `translation_results`)
2. ✅ Backend translation API missing supported languages (mr, gu, de, en)
3. ✅ Backend translation API status check too restrictive (rejects "in-review")
4. 🟡 Queue cards show total pages instead of selected pages (confusing UX)
5. 🟡 Dashboard data integration completely broken (all widgets empty)

#### Task 1.1: Add Auto-Translate Logic After Extraction Completion

Add after line 296 (after updating status to "in-review"):

```python
# ============================================
# AUTO-TRANSLATE PIPELINE INTEGRATION
# ============================================

# Get job metadata from extraction result
job_metadata = extraction_result.get("metadata", {})

# Check if auto-translate is enabled
if job_metadata.get("auto_translate"):
    target_languages = job_metadata.get("target_languages", [])
    source_language = job_metadata.get("source_language", "en")
    priority = job_metadata.get("priority", "normal")

    self.logger.info(
        f"   [AUTO-TRANSLATE] Enabled: Creating {len(target_languages)} translation jobs",
        extra={
            "job_id": job_id,
            "document_id": document_id,
            "target_languages": target_languages
        }
    )

    # Create translation job for each target language
    translation_job_ids = []
    for target_lang in target_languages:
        try:
            # Create translation job record in database
            translation_job_data = {
                "document_id": document_id,
                "extraction_result_id": job_id,
                "source_language": source_language,
                "target_language": target_lang,
                "status": "queued",
                "progress": 0,
                "priority": priority,
                "queued_at": datetime.utcnow().isoformat(),
                "metadata": {
                    "auto_translate": True,
                    "triggered_by": "text_extraction_worker",
                    "extraction_job_id": job_id
                }
            }

            # Insert into translation_results table
            translation_result = self.supabase.table("translation_results") \
                .insert(translation_job_data) \
                .execute()

            translation_job_id = translation_result.data[0]["id"]
            translation_job_ids.append(translation_job_id)

            # Add job to translation_queue via PGMQ
            pgmq_message = {
                "job_id": translation_job_id,
                "document_id": document_id,
                "extraction_result_id": job_id,
                "source_language": source_language,
                "target_language": target_lang,
                "priority": priority
            }

            msg_id = self.pgmq.send("translation_queue", pgmq_message)

            self.logger.info(
                f"      ✓ Created translation job: {target_lang} (msg_id: {msg_id})",
                extra={
                    "translation_job_id": translation_job_id,
                    "target_language": target_lang
                }
            )

        except Exception as e:
            self.logger.error(
                f"      ✗ Failed to create translation job for {target_lang}: {e}",
                extra={
                    "target_language": target_lang,
                    "error": str(e)
                }
            )
            continue

    # Update document's pipeline status
    if translation_job_ids:
        self.supabase.table("documents").update({
            "current_pipeline": "translation",
            "translation_status": "queued",
            "updated_at": datetime.utcnow().isoformat()
        }).eq("id", document_id).execute()

        self.logger.info(
            f"   [AUTO-TRANSLATE] Complete: {len(translation_job_ids)} jobs created",
            extra={
                "job_ids": translation_job_ids
            }
        )
else:
    self.logger.info(
        f"   [AUTO-TRANSLATE] Disabled: Extraction complete, awaiting manual review",
        extra={"job_id": job_id}
    )
```

```typescript
// Add helper to calculate display text
const getPageDisplayText = (job: QueueJob): string => {
  const metadata = job.metadata || {};
  const selectedPages = metadata.selected_pages || [];
  const totalPages = metadata.total_pages || 0;

  if (selectedPages.length === 0 || selectedPages.length === totalPages) {
    return `${totalPages} pages`;
  }

  return `${selectedPages.length} of ${totalPages} pages`;
};
```

#### Task 3.2: Implement Stats Cards

```typescript
// In dashboard-store.ts

fetchStats: async () => {
  // Pending count (queued jobs)
  const { count: pendingCount } = await supabase
    .from('text_extraction_results')
    .select('*', { count: 'exact', head: true })
    .eq('status', 'queued');

  // Processing count (processing + in-review)
  const { count: processingCount } = await supabase
    .from('text_extraction_results')
    .select('*', { count: 'exact', head: true })
    .or('status.eq.processing,status.eq.in-review');

  // Completed today
  const today = new Date().toISOString().split('T')[0];
  const { count: completedToday } = await supabase
    .from('text_extraction_results')
    .select('*', { count: 'exact', head: true })
    .eq('status', 'completed')
    .gte('completed_at', `${today}T00:00:00`);

  set({
    stats: {
      pending: pendingCount || 0,
      processing: processingCount || 0,
      completedToday: completedToday || 0
    }
  });
}
```

#### Task 3.3: Implement System Health Check

```typescript
// In dashboard-store.ts

checkSystemHealth: async () => {
  let supabaseOnline = false;
  let geminiOnline = false;

  // Check Supabase
  try {
    const { error } = await supabase.from('documents').select('id').limit(1);
    supabaseOnline = !error;
  } catch {
    supabaseOnline = false;
  }

  // Check Gemini AI (check if any recent successful jobs)
  try {
    const { data } = await supabase
      .from('text_extraction_results')
      .select('completed_at')
      .eq('status', 'completed')
      .order('completed_at', { ascending: false })
      .limit(1);

    // If we have a job completed in the last hour, assume Gemini is online
    if (data && data.length > 0) {
      const lastCompleted = new Date(data[0].completed_at);
      const oneHourAgo = new Date(Date.now() - 60 * 60 * 1000);
      geminiOnline = lastCompleted > oneHourAgo;
    }
  } catch {
    geminiOnline = false;
  }

  set({
    systemHealth: {
      supabase: supabaseOnline,
      gemini: geminiOnline
    }
  });
}
```

```typescript
// In upload-store.ts, togglePageSelection action (around line 277-289)

togglePageSelection: (fileId: string, pageId: string) =>
  set((state) => {
    const newState = {
      files: state.files.map((file) =>
        file.id === fileId && file.pages
          ? {
              ...file,
              pages: file.pages.map((page) =>
                page.id === pageId ? { ...page, selected: !page.selected } : page
              )
            }
          : file
      )
    };

    // Recalculate cost after updating selection
    setTimeout(() => get().calculateCost(), 0);

    return newState;
  }),
```
