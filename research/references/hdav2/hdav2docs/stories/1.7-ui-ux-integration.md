```tsx
import { useEffect, useState } from 'react';
import { supabase } from '@/lib/supabase';
import { useQuery } from '@tanstack/react-query';

export default function Queue() {
  // Fetch queue status from worker API
  const { data: workerStatus, refetch: refetchWorkerStatus } = useQuery({
    queryKey: ['worker-status'],
    queryFn: async () => {
      const response = await fetch('http://localhost:8000/api/workers/status');
      return response.json();
    },
    refetchInterval: 3000, // Refresh every 3 seconds
  });

  // Fetch all jobs from database
  const { data: jobs, refetch: refetchJobs } = useQuery({
    queryKey: ['extraction-jobs'],
    queryFn: async () => {
      const { data } = await supabase
        .from('text_extraction_results')
        .select('*, documents(*)')
        .order('created_at', { ascending: false })
        .limit(50);
      return data;
    },
    refetchInterval: 3000,
  });

  // Subscribe to real-time updates
  useEffect(() => {
    const channel = supabase
      .channel('extraction-updates')
      .on(
        'postgres_changes',
        {
          event: '*',
          schema: 'public',
          table: 'text_extraction_results',
        },
        (payload) => {
          console.log('Real-time update:', payload);
          refetchJobs();
          refetchWorkerStatus();
        }
      )
      .subscribe();

    return () => {
      supabase.removeChannel(channel);
    };
  }, [refetchJobs, refetchWorkerStatus]);

  // Group jobs by status
  const queuedJobs = jobs?.filter((j) => j.status === 'queued' || j.status === 'queued_manual') || [];
  const processingJobs = jobs?.filter((j) => j.status === 'processing') || [];
  const completedJobs = jobs?.filter((j) => j.status === 'completed') || [];
  const failedJobs = jobs?.filter((j) => j.status === 'failed') || [];

  return (
    <div className="queue-page">
      {/* Worker Status Dashboard */}
      <div className="worker-status-card">
        <h2>Worker Pool Status</h2>
        <div className="stats">
          <div className="stat">
            <span className="label">Active Workers</span>
            <span className="value">{workerStatus?.active_jobs || 0} / {workerStatus?.worker_pool_size || 4}</span>
          </div>
          <div className="stat">
            <span className="label">Queue Depth</span>
            <span className="value">{workerStatus?.queue_depth || 0} jobs</span>
          </div>
          <div className="stat">
            <span className="label">Completed Today</span>
            <span className="value">{workerStatus?.completed_today || 0}</span>
          </div>
        </div>
      </div>

      {/* Job Columns */}
      <div className="job-columns">
        {/* Queued Column */}
        <div className="job-column">
          <h3>Queued ({queuedJobs.length})</h3>
          {queuedJobs.length === 0 ? (
            <div className="empty-state">No jobs queued</div>
          ) : (
            queuedJobs.map((job, index) => (
              <JobCard
                key={job.id}
                job={job}
                position={index + 1}
                showPosition
              />
            ))
          )}
        </div>

        {/* Processing Column */}
        <div className="job-column">
          <h3>Processing ({processingJobs.length})</h3>
          {processingJobs.length === 0 ? (
            <div className="empty-state">No jobs processing</div>
          ) : (
            processingJobs.map((job) => (
              <JobCard key={job.id} job={job} showProgress />
            ))
          )}
        </div>

        {/* Completed Column */}
        <div className="job-column">
          <h3>Completed ({completedJobs.length})</h3>
          {completedJobs.length === 0 ? (
            <div className="empty-state">No completed jobs</div>
          ) : (
            completedJobs.map((job) => (
              <JobCard key={job.id} job={job} showViewButton />
            ))
          )}
        </div>
      </div>
    </div>
  );
}

function JobCard({ job, position, showProgress, showViewButton, showPosition }) {
  return (
    <div className="job-card">
      <div className="job-header">
        <h4>{job.documents.original_name}</h4>
        {showPosition && <span className="position">#{position}</span>}
      </div>
      <div className="job-details">
        <span>{job.documents.page_count} pages</span>
        {showProgress && (
          <div className="progress-bar">
            <div className="progress-fill" style={{ width: `${job.progress}%` }} />
            <span>{job.progress}%</span>
          </div>
        )}
        {showViewButton && (
          <button onClick={() => window.location.href = `/editor/${job.document_id}`}>
            View
          </button>
        )}
      </div>
    </div>
  );
}
```

### Focus Area 2: Upload Page Improvements

**File**: `frontend/src/pages/Upload.tsx`

**Improvements**:

```tsx
import { useState } from 'react';
import { useDropzone } from 'react-dropzone';
import { toast } from 'sonner';

export default function Upload() {
  const [uploadProgress, setUploadProgress] = useState(0);
  const [isUploading, setIsUploading] = useState(false);

  const { getRootProps, getInputProps, isDragActive } = useDropzone({
    accept: { 'application/pdf': ['.pdf'] },
    multiple: true,
    maxSize: 500 * 1024 * 1024, // 500 MB
    onDrop: handleFileDrop,
  });

  async function handleFileDrop(files) {
    if (files.length === 0) {
      toast.error('No valid files selected');
      return;
    }

    setIsUploading(true);

    for (const file of files) {
      try {
        await uploadFile(file);
        toast.success(`${file.name} uploaded successfully`);
      } catch (error) {
        toast.error(`Failed to upload ${file.name}: ${error.message}`);
      }
    }

    setIsUploading(false);
    setUploadProgress(0);
  }

  async function uploadFile(file) {
    const formData = new FormData();
    formData.append('file', file);

    const xhr = new XMLHttpRequest();

    return new Promise((resolve, reject) => {
      xhr.upload.addEventListener('progress', (e) => {
        if (e.lengthComputable) {
          const progress = Math.round((e.loaded / e.total) * 100);
          setUploadProgress(progress);
        }
      });

      xhr.addEventListener('load', () => {
        if (xhr.status === 200) {
          resolve(JSON.parse(xhr.responseText));
        } else {
          reject(new Error(xhr.statusText));
        }
      });

      xhr.addEventListener('error', () => reject(new Error('Upload failed')));

      xhr.open('POST', 'http://localhost:8000/api/documents/upload');
      xhr.send(formData);
    });
  }

  return (
    <div className="upload-page">
      <h1>Upload Documents</h1>

      {/* Drag-Drop Zone */}
      <div
        {...getRootProps()}
        className={`dropzone ${isDragActive ? 'active' : ''} ${isUploading ? 'uploading' : ''}`}
      >
        <input {...getInputProps()} />
        {isUploading ? (
          <div className="upload-progress">
            <div className="spinner" />
            <p>Uploading... {uploadProgress}%</p>
            <div className="progress-bar">
              <div className="fill" style={{ width: `${uploadProgress}%` }} />
            </div>
          </div>
        ) : isDragActive ? (
          <p>Drop files here...</p>
        ) : (
          <div>
            <p>Drag & drop PDF files here, or click to browse</p>
            <span className="hint">Max file size: 500 MB</span>
          </div>
        )}
      </div>

      {/* Action Buttons */}
      <div className="actions">
        <button className="btn-primary" disabled={isUploading}>
          Process Now
        </button>
        <button className="btn-secondary" disabled={isUploading}>
          Add to Queue
        </button>
      </div>
    </div>
  );
}
```

```tsx
      } else if (e.key === 'ArrowRight') {
        setCurrentPage((prev) => Math.min(totalPages, prev + 1));
      }
    }

    window.addEventListener('keydown', handleKeyPress);
    return () => window.removeEventListener('keydown', handleKeyPress);
  }, []);

  // Copy text to clipboard
  function copyPageText() {
    const pageText = getCurrentPageText();
    navigator.clipboard.writeText(pageText);
    toast.success('Page text copied to clipboard');
  }

  if (loading) {
    return (
      <div className="editor-loading">
        <div className="skeleton-header" />
        <div className="skeleton-content" />
      </div>
    );
  }
```
