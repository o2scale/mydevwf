## Critical Requirements

### 1. Page Marker Format (BLOCKING ISSUE)

Frontend editor **depends** on exact format from `FRONTEND_BACKEND_GAP_ANALYSIS.md` Section 5.3:

```
# [Page 1]
First line of page 1 text
Second line of page 1 text

# [Page 2]
First line of page 2 text
Second line of page 2 text
```

**Regex Used by Frontend**: `/^#\s*\[Page\s+(\d+)\]$/i`

**What Breaks if Wrong**:
- Editor cannot assign page numbers to lines
- Total page count shows as 0
- Page navigation doesn't work
- Editor displays single blob of text

**Implementation**: Backend MUST insert these markers after Vertex AI extraction.

---

### 2. Token-Based Batching (from ARCHITECTURE_DECISIONS.md Decision 7)

**Vertex AI Limits**:
- Max input: 1,000,000 tokens
- Safe limit per batch: 900,000 tokens (10% buffer)

**Batch Strategy**:
- Digital PDFs: ~500-800 tokens per page → 50-100 pages per batch
- Scanned PDFs: ~300-500 tokens per page → 20-30 pages per batch

**Why Token-Based (not page-based)**:
- Pages vary in content density
- Prevent hitting API limits mid-batch
- Optimize for cost and performance

```python
from fastapi import APIRouter, HTTPException
from pydantic import BaseModel
from typing import Optional, List
from supabase import create_client
import os
import uuid
from datetime import datetime

router = APIRouter(prefix="/api/text-extraction", tags=["text-extraction"])

supabase = create_client(
    os.getenv("SUPABASE_URL"),
    os.getenv("SUPABASE_SERVICE_KEY")
)

class ExtractionRequest(BaseModel):
    auto_translate: bool = False
    target_languages: Optional[List[str]] = []
    selected_pages: Optional[List[int]] = None

@router.post("/extract-with-batching/{document_id}")
async def start_extraction_with_batching(
    document_id: str,
    request: ExtractionRequest
):
    """
    Start text extraction job with intelligent batching

    Args:
        document_id: UUID of uploaded document
        request: Extraction configuration

    Returns:
        Job ID and batch information
    """

    # ========================================================================
    # STEP 1: Validate Document Exists
    # ========================================================================
    doc = supabase.table("documents").select("*").eq("id", document_id).execute()

    if not doc.data:
        raise HTTPException(status_code=404, detail="Document not found")

    document = doc.data[0]

    if document["status"] not in ["uploaded", "completed"]:
        raise HTTPException(
            status_code=400,
            detail=f"Document status must be 'uploaded' or 'completed', got '{document['status']}'"
        )

    # ========================================================================
    # STEP 2: Create Extraction Job Record
    # ========================================================================
    job_id = str(uuid.uuid4())

    job_data = {
        "id": job_id,
        "document_id": document_id,
        "status": "queued",
        "progress": 0,
        "service_used": "vertex-ai-gemini-2.5-pro",
        "selected_pages": request.selected_pages
    }

    supabase.table("text_extraction_results").insert(job_data).execute()

    # ========================================================================
    # STEP 3: Calculate Batch Strategy
    # ========================================================================
    from api.services.batch_processor import calculate_batches

    total_pages = document["page_count"]
    pages_to_process = request.selected_pages if request.selected_pages else list(range(1, total_pages + 1))

    batch_info = calculate_batches(
        document_id=document_id,
        job_id=job_id,
        total_pages=total_pages,
        selected_pages=pages_to_process
    )

    # ========================================================================
    # STEP 4: Send Message to pgmq Queue
    # ========================================================================
    import psycopg2
    import json

    conn = psycopg2.connect(os.getenv("DATABASE_URL"))
    cursor = conn.cursor()

    queue_message = {
        "job_id": job_id,
        "document_id": document_id,
        "batch_count": batch_info["total_batches"],
        "priority": 8 if total_pages <= 10 else 5,
        "selected_pages": pages_to_process
    }

    cursor.execute(
        "SELECT pgmq.send(%s, %s::jsonb);",
        ("text_extraction_queue", json.dumps(queue_message))
    )
    conn.commit()
    cursor.close()
    conn.close()

    # ========================================================================
    # STEP 5: Update Document Status
    # ========================================================================
    supabase.table("documents").update({"status": "processing"}).eq("id", document_id).execute()

    # ========================================================================
    # STEP 6: Return Response
    # ========================================================================
    return {
        "job_id": job_id,
        "document_id": document_id,
        "status": "queued",
        "message": "Text extraction job queued successfully",
        "estimated_time": len(pages_to_process) * 2,  # 2 seconds per page estimate
        "batch_info": batch_info
    }
```

### Step 2: Implement Batch Calculation Service

**File**: `backend/api/services/batch_processor.py`

```python
from typing import List, Dict

def estimate_tokens(page_count: int, document_type: str = "digital") -> int:
    """
    Estimate token count for PDF pages

    Token estimation rules:
    - Digital text PDFs: ~500-800 tokens per page
    - Scanned/image PDFs: ~300-500 tokens per page
    - Add 20% buffer for safety
    """
    if document_type == "digital":
        tokens_per_page = 650  # Average
    else:  # scanned
        tokens_per_page = 400  # Average

    estimated = page_count * tokens_per_page
    buffered = int(estimated * 1.2)  # Add 20% buffer

    return buffered

def calculate_batch_size(total_pages: int, estimated_tokens: int, document_type: str = "digital") -> int:
    """
    Calculate optimal batch size based on token limits

    Vertex AI Gemini 2.5 Pro limits:
    - Max input tokens: 1,000,000
    - Safe limit per batch: 900,000 (leaving 100K buffer)
    """
    SAFE_TOKEN_LIMIT = 900_000

    tokens_per_page = estimated_tokens / total_pages
    max_pages_per_batch = int(SAFE_TOKEN_LIMIT / tokens_per_page)

    # Apply document-type specific limits
    if document_type == "digital":
        batch_size = min(max_pages_per_batch, 100)
    else:  # scanned
        batch_size = min(max_pages_per_batch, 30)

    return max(batch_size, 1)  # At least 1 page per batch

def calculate_batches(
    document_id: str,
    job_id: str,
    total_pages: int,
    selected_pages: List[int],
    document_type: str = "digital"
) -> Dict:
    """
    Create batch records for a document

    Returns:
        Batch information dictionary
    """
    from supabase import create_client
    import os

    supabase = create_client(
        os.getenv("SUPABASE_URL"),
        os.getenv("SUPABASE_SERVICE_KEY")
    )

    # Estimate tokens
    estimated_tokens = estimate_tokens(len(selected_pages), document_type)

    # Calculate batch size
    batch_size = calculate_batch_size(len(selected_pages), estimated_tokens, document_type)

    # Create batches
    batches = []
    for i in range(0, len(selected_pages), batch_size):
        batch_pages = selected_pages[i:i + batch_size]
        batch_number = (i // batch_size) + 1
        total_batches = (len(selected_pages) + batch_size - 1) // batch_size

        batch = {
            "document_id": document_id,
            "extraction_job_id": job_id,
            "batch_number": batch_number,
            "total_batches": total_batches,
            "start_page": batch_pages[0],
            "end_page": batch_pages[-1],
            "page_count": len(batch_pages),
            "estimated_tokens": int(estimated_tokens * len(batch_pages) / len(selected_pages)),
            "status": "pending"
        }
        batches.append(batch)

    # Insert into database
    if batches:
        supabase.table("processing_batches").insert(batches).execute()

    return {
        "total_pages": total_pages,
        "selected_pages": len(selected_pages),
        "total_batches": len(batches),
        "estimated_tokens": estimated_tokens,
        "batch_strategy": "token-based",
        "batches": [
            {
                "batch_number": b["batch_number"],
                "start_page": b["start_page"],
                "end_page": b["end_page"],
                "page_count": b["page_count"]
            }
            for b in batches
        ]
    }
```

```python
@router.get("/status/{job_id}")
async def get_extraction_status(job_id: str):
    """
    Get extraction job status and progress

    Args:
        job_id: UUID of extraction job

    Returns:
        Job status, progress, and batch information
    """

    # Get job status
    job = supabase.table("text_extraction_results").select("*").eq("id", job_id).execute()

    if not job.data:
        raise HTTPException(status_code=404, detail="Job not found")

    job_data = job.data[0]

    # Get batch progress
    batches = supabase.table("processing_batches").select("*").eq("extraction_job_id", job_id).execute()

    if batches.data:
        completed_batches = len([b for b in batches.data if b["status"] == "completed"])
        total_batches = len(batches.data)
        current_batch = completed_batches + 1 if completed_batches < total_batches else total_batches

        # Calculate current page (rough estimate)
        if completed_batches < total_batches:
            current_page = batches.data[completed_batches]["end_page"] if completed_batches > 0 else 0
        else:
            current_page = job_data["page_count"]
    else:
        completed_batches = 0
        total_batches = 0
        current_batch = 0
        current_page = 0

    # Calculate progress
    progress = job_data["progress"]

    # Build response based on status
    response = {
        "job_id": job_id,
        "document_id": job_data["document_id"],
        "status": job_data["status"],
        "progress": progress
    }

    if job_data["status"] == "processing":
        response.update({
            "current_batch": current_batch,
            "total_batches": total_batches,
            "current_page": current_page,
            "total_pages": job_data["page_count"],
            "message": f"Processing batch {current_batch} of {total_batches}"
        })
    elif job_data["status"] == "completed":
        response.update({
            "total_pages": job_data["page_count"],
            "total_batches": total_batches,
            "token_count": job_data["token_count"],
            "processing_time": job_data["processing_time"],
            "confidence_score": float(job_data["confidence_score"]) if job_data["confidence_score"] else None,
            "completed_at": job_data["completed_at"],
            "message": "Text extraction completed successfully"
        })
    elif job_data["status"] == "failed":
        response.update({
            "error_message": job_data["error_message"],
            "retry_count": job_data["retry_count"],
            "message": "Text extraction failed"
        })
    else:
        response["message"] = "Text extraction queued"

    return response
```

### Step 4: Implement Text Extraction Worker

**File**: `backend/workers/text_extraction_worker.py`

```python
import asyncio
import psycopg2
import json
import os
from dotenv import load_dotenv
from supabase import create_client
import vertexai
from vertexai.generative_models import GenerativeModel, Part
from PyPDF2 import PdfReader
import io
from PIL import Image
import pdf2image
from datetime import datetime

load_dotenv()

class TextExtractionWorker:
    def __init__(self):
        # Database connection for pgmq
        self.db_conn = psycopg2.connect(os.getenv("DATABASE_URL"))

        # Supabase client
        self.supabase = create_client(
            os.getenv("SUPABASE_URL"),
            os.getenv("SUPABASE_SERVICE_KEY")
        )

        # Vertex AI client
        vertexai.init(
            project=os.getenv("GOOGLE_CLOUD_PROJECT"),
            location=os.getenv("VERTEX_AI_LOCATION", "us-central1")
        )
        self.vertex_model = GenerativeModel(os.getenv("VERTEX_AI_MODEL", "gemini-2.0-flash-exp"))

    async def run(self):
        """Main worker loop"""
        print("🚀 Text Extraction Worker started")
        print(f"   Polling queue: text_extraction_queue")
        print(f"   Model: {os.getenv('VERTEX_AI_MODEL')}")

        while True:
            try:
                # Read message from queue (5 minute visibility timeout)
                cursor = self.db_conn.cursor()
                cursor.execute("""
                    SELECT * FROM pgmq.read('text_extraction_queue', 300, 1);
                """)
                messages = cursor.fetchall()
                cursor.close()

                if not messages:
                    await asyncio.sleep(2)  # Poll every 2 seconds
                    continue

                msg = messages[0]
                msg_id = msg[0]
                job_data = msg[1]  # JSONB message content

                print(f"\n📥 Received job: {job_data['job_id']}")

                # Process job
                await self.process_extraction_job(job_data)

                # Archive message
                cursor = self.db_conn.cursor()
                cursor.execute("SELECT pgmq.archive('text_extraction_queue', %s);", (msg_id,))
                self.db_conn.commit()
                cursor.close()

                print(f"✅ Job {job_data['job_id']} archived from queue")

            except Exception as e:
                print(f"❌ Worker error: {e}")
                await asyncio.sleep(5)

    async def process_extraction_job(self, job_data: dict):
        """Process a single extraction job"""
        job_id = job_data["job_id"]
        document_id = job_data["document_id"]

        try:
            # Update job status to processing
            self.supabase.table("text_extraction_results").update({
                "status": "processing",
                "progress": 0
            }).eq("id", job_id).execute()

            # Get document info
            doc = self.supabase.table("documents").select("*").eq("id", document_id).single().execute()
            document = doc.data

            # Download PDF from storage
            pdf_bytes = self.supabase.storage.from_("documents").download(document["storage_path"])

            # Get batches for this job
            batches = self.supabase.table("processing_batches").select("*").eq(
                "extraction_job_id", job_id
            ).order("batch_number").execute()

            total_batches = len(batches.data)
            batch_results = []

            # Process each batch
            for i, batch in enumerate(batches.data):
                print(f"   Processing batch {batch['batch_number']} of {total_batches}")

                # Update batch status
                self.supabase.table("processing_batches").update({
                    "status": "processing"
                }).eq("id", batch["id"]).execute()

                # Extract text for this batch
                batch_text = await self.extract_batch(
                    pdf_bytes,
                    batch["start_page"],
                    batch["end_page"]
                )

                # Update batch with results
                self.supabase.table("processing_batches").update({
                    "status": "completed",
                    "extracted_text": batch_text,
                    "completed_at": datetime.now().isoformat()
                }).eq("id", batch["id"]).execute()

                batch_results.append({
                    "batch_number": batch["batch_number"],
                    "start_page": batch["start_page"],
                    "end_page": batch["end_page"],
                    "extracted_text": batch_text
                })

                # Update overall progress
                progress = int(((i + 1) / total_batches) * 100)
                self.supabase.table("text_extraction_results").update({
                    "progress": progress
                }).eq("id", job_id).execute()

                print(f"   ✓ Batch {batch['batch_number']} complete ({progress}%)")

            # Combine results with page markers
            full_text = self.insert_page_markers(batch_results)

            # Update job as completed
            self.supabase.table("text_extraction_results").update({
                "status": "completed",
                "progress": 100,
                "extracted_text": full_text,
                "completed_at": datetime.now().isoformat(),
                "page_count": len(job_data.get("selected_pages", []))
            }).eq("id", job_id).execute()

            # Update document status
            self.supabase.table("documents").update({
                "status": "completed"
            }).eq("id", document_id).execute()

            print(f"✅ Job {job_id} completed successfully")

        except Exception as e:
            # Handle failure
            self.supabase.table("text_extraction_results").update({
                "status": "failed",
                "error_message": str(e)
            }).eq("id", job_id).execute()

            print(f"❌ Job {job_id} failed: {e}")

    async def extract_batch(self, pdf_bytes: bytes, start_page: int, end_page: int) -> str:
        """Extract text from a batch of pages using Vertex AI"""
        print(f"      Calling Vertex AI for pages {start_page}-{end_page}")

        # Convert PDF pages to images
        images = pdf2image.convert_from_bytes(
            pdf_bytes,
            first_page=start_page,
            last_page=end_page
        )

        # Prepare prompt
        prompt = f"""Extract all text from these document pages ({start_page} to {end_page}).

Requirements:
- Preserve original text layout and structure
- Include all text, even if partially visible
- Maintain paragraph breaks and formatting
- If text is unclear, provide best-effort transcription

Return the extracted text only, without any additional commentary."""

        # Convert images to Vertex AI Part objects
        parts = [prompt]
        for img in images:
            img_bytes = io.BytesIO()
            img.save(img_bytes, format="PNG")
            parts.append(Part.from_data(img_bytes.getvalue(), mime_type="image/png"))

        # Call Vertex AI
        response = await self.vertex_model.generate_content_async(parts)

        return response.text

    def insert_page_markers(self, batch_results: list) -> str:
        """
        Combine batch results and insert page markers

        CRITICAL: Page marker format MUST be exact: # [Page N]
        """
        combined_text = []

        for batch in sorted(batch_results, key=lambda x: x["batch_number"]):
            start_page = batch["start_page"]
            end_page = batch["end_page"]
            batch_text = batch["extracted_text"]

            # Split batch text into estimated pages
            lines = batch_text.split("\n")
            page_count = end_page - start_page + 1
            lines_per_page = max(len(lines) // page_count, 1)

            for i in range(page_count):
                page_num = start_page + i
                start_line = i * lines_per_page
                end_line = (i + 1) * lines_per_page if i < page_count - 1 else len(lines)
                page_text = "\n".join(lines[start_line:end_line])

                # Insert page marker (EXACT FORMAT)
                combined_text.append(f"# [Page {page_num}]")
                combined_text.append(page_text.strip())
                combined_text.append("")  # Blank line between pages

        return "\n".join(combined_text)


# Run worker
if __name__ == "__main__":
    worker = TextExtractionWorker()
    asyncio.run(worker.run())
```
