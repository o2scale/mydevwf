**Epic**: Epic 1 - Core Translation System
**Priority**: P1 (User Experience)
**Estimate**: 1-2 hours
**Status**: 📋 PLANNED (2025-10-03)
**Depends On**: Story 1.5 (Parallel Worker Architecture)

---

## User Story

**As a** User
**I want** to add documents to the processing queue without immediate processing
**So that** I can batch upload multiple documents and let them process when workers are available

---

## Context

**Current Behavior**:
- "Process Now" button triggers immediate text extraction
- No "Add to Queue" functionality (button exists but not wired)
- Users must wait for each document to start processing

**Desired Behavior**:
- "Process Now" → Starts extraction immediately if workers available
- "Add to Queue" → Adds to queue, processes when worker becomes free
- Bulk upload workflow: Add 10 docs to queue, watch 4 process concurrently

**Queue Page**: http://localhost:8080/queue currently shows mock data

**Add helper functions**:

```python
def get_queue_position(job_id: str) -> int:
    """Get position of job in queue"""
    import psycopg2

    conn = psycopg2.connect(os.getenv("DATABASE_URL"))
    cursor = conn.cursor()

    cursor.execute("""
        SELECT COUNT(*)
        FROM pgmq.text_extraction_queue
        WHERE msg_id < (
            SELECT msg_id
            FROM pgmq.text_extraction_queue
            WHERE message->>'job_id' = %s
        );
    """, (job_id,))

    position = cursor.fetchone()[0] + 1
    cursor.close()
    conn.close()

    return position


def calculate_wait_time() -> int:
    """Calculate estimated wait time in seconds"""
    import psycopg2

    conn = psycopg2.connect(os.getenv("DATABASE_URL"))
    cursor = conn.cursor()

    # Get queue depth
    cursor.execute("SELECT COUNT(*) FROM pgmq.text_extraction_queue;")
    queue_depth = cursor.fetchone()[0]

    # Get active jobs
    cursor.execute("""
        SELECT COUNT(*)
        FROM text_extraction_results
        WHERE status = 'processing';
    """)
    active_jobs = cursor.fetchone()[0]

    cursor.close()
    conn.close()

    # Calculation
    worker_pool_size = int(os.getenv("WORKER_POOL_SIZE", "4"))
    available_workers = max(worker_pool_size - active_jobs, 0)

    if available_workers > 0:
        return 0  # Will start immediately
    else:
        # Estimate: 60 seconds per job average
        jobs_ahead = queue_depth - available_workers
        return jobs_ahead * 60
```

### Step 2: Update Frontend Upload Page

**File**: `frontend/src/pages/Upload.tsx`

**Add "Add to Queue" button handler**:

```tsx
const handleAddToQueue = async () => {
  if (!uploadedDocument) {
    toast.error('Please upload a document first');
    return;
  }

  try {
    setIsProcessing(true);

    const response = await fetch(
      `http://localhost:8000/api/text-extraction/extract-with-batching/${uploadedDocument.id}`,
      {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          auto_translate: false,
          selected_pages: null, // Process all pages
          queue_only: true, // NEW FLAG
        }),
      }
    );

    if (!response.ok) {
      throw new Error('Failed to add document to queue');
    }

    const data = await response.json();

    // Show success message with queue position
    const queuePosition = data.queue_position;
    const waitTime = data.estimated_wait_time;

    if (queuePosition) {
      toast.success(
        `Document added to queue (position #${queuePosition}, ~${Math.ceil(waitTime / 60)} min wait)`
      );
    } else {
      toast.success('Document added to queue');
    }

    // Navigate to queue page
    navigate('/queue');

  } catch (error) {
    console.error('Error adding to queue:', error);
    toast.error('Failed to add document to queue');
  } finally {
    setIsProcessing(false);
  }
};
```

**Update JSX** (locate the button section):


### Step 3: Update Queue Monitoring Page

**File**: `frontend/src/pages/Queue.tsx`

**Add real queue data fetching**:

```tsx
import { useQuery } from '@tanstack/react-query';

// Fetch queue status
const { data: queueData, refetch } = useQuery({
  queryKey: ['queue-status'],
  queryFn: async () => {
    const response = await fetch('http://localhost:8000/api/workers/status');
    return response.json();
  },
  refetchInterval: 3000, // Refresh every 3 seconds
});

// Fetch all jobs
const { data: jobs } = useQuery({
  queryKey: ['extraction-jobs'],
  queryFn: async () => {
    const response = await supabase
      .from('text_extraction_results')
      .select('*, documents(*)')
      .order('created_at', { ascending: false })
      .limit(50);
    return response.data;
  },
  refetchInterval: 3000,
});

// Group jobs by status
const queuedJobs = jobs?.filter(j => j.status === 'queued_manual') || [];
const processingJobs = jobs?.filter(j => j.status === 'processing') || [];
const completedJobs = jobs?.filter(j => j.status === 'completed') || [];
```
