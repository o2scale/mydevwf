## Context

**Problem Identified** (from devnotes 2025-10-04 12:42:00):
- Frontend expects `translation_results` table, but database has `translations` table
- Missing review workflow columns (`review_status`, `reviewer_id`, etc.)
- Old status values (`text-processing`, `queued_manual`) don't match new unified flow
- No pipeline tracking in `documents` table
- Need to support one-row-per-language approach for granular translation tracking

```

```

**Desired State**:
```
✅ documents - Enhanced with pipeline tracking columns
```

## Migration Strategy

**Note**: Starting fresh - no backup needed. All existing test data will be cleared.

### Phase 1: Update `text_extraction_results` Table

```sql
-- ============================================
-- PHASE 1: UPDATE text_extraction_results TABLE
-- ============================================

-- Add review workflow columns
ALTER TABLE text_extraction_results
ADD COLUMN IF NOT EXISTS review_status VARCHAR(50) DEFAULT 'pending',
ADD COLUMN IF NOT EXISTS reviewer_id UUID,
ADD COLUMN IF NOT EXISTS reviewed_at TIMESTAMP WITH TIME ZONE,
ADD COLUMN IF NOT EXISTS review_notes TEXT,
ADD COLUMN IF NOT EXISTS started_at TIMESTAMP WITH TIME ZONE,
ADD COLUMN IF NOT EXISTS processing_duration_ms INTEGER;

-- Add check constraint for review_status
ALTER TABLE text_extraction_results
ADD CONSTRAINT check_review_status CHECK (review_status IN ('pending', 'approved', 'rejected'));

-- Add check constraint for status (unified values)
ALTER TABLE text_extraction_results
ADD CONSTRAINT check_status CHECK (status IN ('queued', 'processing', 'in-review', 'completed', 'failed'));

-- Add foreign key for reviewer_id (will be used in Epic 2)
ALTER TABLE text_extraction_results
ADD CONSTRAINT fk_reviewer FOREIGN KEY (reviewer_id) REFERENCES auth.users(id) ON DELETE SET NULL;

-- Migrate old status values to new unified format
UPDATE text_extraction_results
SET status = CASE
    WHEN status = 'text-processing' THEN 'processing'
    WHEN status = 'queued_manual' THEN 'queued'
    ELSE status
END
WHERE status IN ('text-processing', 'queued_manual');

-- Add index for review filtering
CREATE INDEX IF NOT EXISTS idx_text_extraction_review_status ON text_extraction_results(review_status);
CREATE INDEX IF NOT EXISTS idx_text_extraction_reviewer_id ON text_extraction_results(reviewer_id) WHERE reviewer_id IS NOT NULL;

-- Verify migration
SELECT DISTINCT status, COUNT(*) as count
FROM text_extraction_results
GROUP BY status
ORDER BY count DESC;
```

### Phase 2: Create `translation_results` Table

**Approach**: Rename existing `translations` table to `translation_results` and add missing columns.

```sql
-- ============================================
-- PHASE 2A: RENAME translations TO translation_results
-- ============================================

-- Check if translations table exists and rename it
ALTER TABLE translations RENAME TO translation_results;

-- ============================================
-- PHASE 2B: ADD MISSING COLUMNS TO translation_results
-- ============================================

-- Add core columns (if missing)
ALTER TABLE translation_results
ADD COLUMN IF NOT EXISTS id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
ADD COLUMN IF NOT EXISTS document_id UUID NOT NULL,
ADD COLUMN IF NOT EXISTS target_language VARCHAR(10) NOT NULL,
ADD COLUMN IF NOT EXISTS source_language VARCHAR(10) DEFAULT 'en',

-- Status tracking (unified 4-stage flow)
ADD COLUMN IF NOT EXISTS status VARCHAR(50) NOT NULL DEFAULT 'queued',
ADD COLUMN IF NOT EXISTS progress INTEGER DEFAULT 0,

-- Priority and processing control
ADD COLUMN IF NOT EXISTS priority INTEGER DEFAULT 5,
ADD COLUMN IF NOT EXISTS process_immediately BOOLEAN DEFAULT false,

-- Translation content
ADD COLUMN IF NOT EXISTS translated_text TEXT,
ADD COLUMN IF NOT EXISTS translation_metadata JSONB,
ADD COLUMN IF NOT EXISTS character_count INTEGER,
ADD COLUMN IF NOT EXISTS word_count INTEGER,

-- Review workflow (Epic 2+)
ADD COLUMN IF NOT EXISTS review_status VARCHAR(50) DEFAULT 'pending',
ADD COLUMN IF NOT EXISTS reviewer_id UUID,
ADD COLUMN IF NOT EXISTS reviewed_at TIMESTAMP WITH TIME ZONE,
ADD COLUMN IF NOT EXISTS review_notes TEXT,

-- Timestamps
ADD COLUMN IF NOT EXISTS created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
ADD COLUMN IF NOT EXISTS updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
ADD COLUMN IF NOT EXISTS queued_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
ADD COLUMN IF NOT EXISTS started_at TIMESTAMP WITH TIME ZONE,
ADD COLUMN IF NOT EXISTS completed_at TIMESTAMP WITH TIME ZONE,

-- Error handling
ADD COLUMN IF NOT EXISTS error_message TEXT,
ADD COLUMN IF NOT EXISTS retry_count INTEGER DEFAULT 0,

-- Performance tracking
ADD COLUMN IF NOT EXISTS processing_duration_ms INTEGER,

-- Multi-tenancy (Epic 2+)
ADD COLUMN IF NOT EXISTS tenant_id UUID;

-- ============================================
-- PHASE 2C: ADD CONSTRAINTS
-- ============================================

-- Add check constraints (using DO blocks for IF NOT EXISTS support)
DO $$
BEGIN
    -- Check constraint for status
    IF NOT EXISTS (
        SELECT 1 FROM pg_constraint WHERE conname = 'check_translation_status'
    ) THEN
        ALTER TABLE translation_results
        ADD CONSTRAINT check_translation_status CHECK (status IN ('queued', 'processing', 'in-review', 'completed', 'failed'));
    END IF;

    -- Check constraint for review_status
    IF NOT EXISTS (
        SELECT 1 FROM pg_constraint WHERE conname = 'check_translation_review_status'
    ) THEN
        ALTER TABLE translation_results
        ADD CONSTRAINT check_translation_review_status CHECK (review_status IN ('pending', 'approved', 'rejected'));
    END IF;

    -- Check constraint for priority
    IF NOT EXISTS (
        SELECT 1 FROM pg_constraint WHERE conname = 'check_translation_priority'
    ) THEN
        ALTER TABLE translation_results
        ADD CONSTRAINT check_translation_priority CHECK (priority >= 1 AND priority <= 10);
    END IF;

    -- Check constraint for progress
    IF NOT EXISTS (
        SELECT 1 FROM pg_constraint WHERE conname = 'check_translation_progress'
    ) THEN
        ALTER TABLE translation_results
        ADD CONSTRAINT check_translation_progress CHECK (progress >= 0 AND progress <= 100);
    END IF;
END $$;

-- Add foreign keys (using DO blocks for IF NOT EXISTS support)
DO $$
BEGIN
    -- Foreign key for document_id
    IF NOT EXISTS (
        SELECT 1 FROM pg_constraint WHERE conname = 'fk_translation_document'
    ) THEN
        ALTER TABLE translation_results
        ADD CONSTRAINT fk_translation_document FOREIGN KEY (document_id) REFERENCES documents(id) ON DELETE CASCADE;
    END IF;

    -- Foreign key for reviewer_id
    IF NOT EXISTS (
        SELECT 1 FROM pg_constraint WHERE conname = 'fk_translation_reviewer'
    ) THEN
        ALTER TABLE translation_results
        ADD CONSTRAINT fk_translation_reviewer FOREIGN KEY (reviewer_id) REFERENCES auth.users(id) ON DELETE SET NULL;
    END IF;
END $$;

-- Add unique constraint (using DO block for IF NOT EXISTS support)
DO $$
BEGIN
    IF NOT EXISTS (
        SELECT 1 FROM pg_constraint WHERE conname = 'unique_doc_language'
    ) THEN
        ALTER TABLE translation_results
        ADD CONSTRAINT unique_doc_language UNIQUE (document_id, target_language);
    END IF;
END $$;

-- ============================================
-- PHASE 2D: CREATE INDEXES
-- ============================================

CREATE INDEX IF NOT EXISTS idx_translation_results_document_id ON translation_results(document_id);
CREATE INDEX IF NOT EXISTS idx_translation_results_status ON translation_results(status);
CREATE INDEX IF NOT EXISTS idx_translation_results_target_language ON translation_results(target_language);
CREATE INDEX IF NOT EXISTS idx_translation_results_created_at ON translation_results(created_at DESC);
CREATE INDEX IF NOT EXISTS idx_translation_results_doc_status ON translation_results(document_id, status);
CREATE INDEX IF NOT EXISTS idx_translation_results_queue_priority ON translation_results(status, priority DESC, created_at ASC);
CREATE INDEX IF NOT EXISTS idx_translation_results_reviewer ON translation_results(reviewer_id) WHERE reviewer_id IS NOT NULL;

-- ============================================
-- PHASE 2E: CREATE AUTO-UPDATE TRIGGER
-- ============================================

-- Create trigger function (if not exists)
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Create trigger on translation_results
DROP TRIGGER IF EXISTS update_translation_results_updated_at ON translation_results;
CREATE TRIGGER update_translation_results_updated_at
    BEFORE UPDATE ON translation_results
    FOR EACH ROW
    EXECUTE FUNCTION update_updated_at_column();

-- Verify table structure
SELECT column_name, data_type, is_nullable, column_default
FROM information_schema.columns
WHERE table_name = 'translation_results'
ORDER BY ordinal_position;
```

### Phase 3: Update `documents` Table

```sql
-- ============================================
-- PHASE 3: ADD PIPELINE TRACKING TO documents
-- ============================================

-- Add pipeline tracking columns
ALTER TABLE documents
ADD COLUMN IF NOT EXISTS current_pipeline VARCHAR(50) DEFAULT 'extraction',
ADD COLUMN IF NOT EXISTS extraction_status VARCHAR(50),
ADD COLUMN IF NOT EXISTS translation_status VARCHAR(50),
ADD COLUMN IF NOT EXISTS extraction_job_id UUID,
ADD COLUMN IF NOT EXISTS translation_job_id UUID;

-- Add check constraint for current_pipeline
ALTER TABLE documents
ADD CONSTRAINT IF NOT EXISTS check_current_pipeline CHECK (current_pipeline IN ('extraction', 'translation', 'completed'));

-- Add foreign keys (optional - for easy navigation)
-- Note: These are "soft" foreign keys since job_id changes over time
-- Commented out for now, can enable if needed:
-- ALTER TABLE documents
-- ADD CONSTRAINT IF NOT EXISTS fk_extraction_job FOREIGN KEY (extraction_job_id) REFERENCES text_extraction_results(id) ON DELETE SET NULL;

-- Create indexes for quick pipeline filtering
CREATE INDEX IF NOT EXISTS idx_documents_current_pipeline ON documents(current_pipeline);
CREATE INDEX IF NOT EXISTS idx_documents_extraction_status ON documents(extraction_status);
CREATE INDEX IF NOT EXISTS idx_documents_translation_status ON documents(translation_status);

-- Populate extraction_status from text_extraction_results (for existing data)
UPDATE documents d
SET
    extraction_status = ter.status,
    extraction_job_id = ter.id
FROM text_extraction_results ter
WHERE d.id = ter.document_id
AND ter.created_at = (
    SELECT MAX(created_at)
    FROM text_extraction_results
    WHERE document_id = d.id
);

-- Verify update
SELECT
    current_pipeline,
    extraction_status,
    COUNT(*) as count
FROM documents
GROUP BY current_pipeline, extraction_status;
```

### Phase 4: Verify Migration

```sql
-- ============================================
-- PHASE 4: VERIFICATION QUERIES
-- ============================================

-- 1. Verify all tables exist
SELECT table_name
FROM information_schema.tables
WHERE table_schema = 'public'
  AND table_name IN ('documents', 'text_extraction_results', 'translation_results')
ORDER BY table_name;

-- 2. Verify status value migration
SELECT 'text_extraction_results' as table_name, status, COUNT(*) as count
FROM text_extraction_results
GROUP BY status
UNION ALL
SELECT 'translation_results', status, COUNT(*)
FROM translation_results
GROUP BY status
ORDER BY table_name, count DESC;

-- 3. Verify foreign keys
SELECT
    tc.table_name,
    kcu.column_name,
    ccu.table_name AS foreign_table_name,
    ccu.column_name AS foreign_column_name,
    rc.delete_rule
FROM information_schema.table_constraints AS tc
JOIN information_schema.key_column_usage AS kcu
    ON tc.constraint_name = kcu.constraint_name
JOIN information_schema.constraint_column_usage AS ccu
    ON ccu.constraint_name = tc.constraint_name
JOIN information_schema.referential_constraints AS rc
    ON tc.constraint_name = rc.constraint_name
WHERE tc.constraint_type = 'FOREIGN KEY'
    AND tc.table_name IN ('text_extraction_results', 'translation_results')
ORDER BY tc.table_name, kcu.column_name;

-- 4. Verify indexes
SELECT
    tablename,
    indexname,
    indexdef
FROM pg_indexes
WHERE tablename IN ('documents', 'text_extraction_results', 'translation_results')
ORDER BY tablename, indexname;

-- 5. Verify unique constraints
SELECT
    tc.table_name,
    tc.constraint_name,
    kcu.column_name
FROM information_schema.table_constraints tc
JOIN information_schema.key_column_usage kcu
    ON tc.constraint_name = kcu.constraint_name
WHERE tc.constraint_type = 'UNIQUE'
    AND tc.table_name IN ('documents', 'text_extraction_results', 'translation_results')
ORDER BY tc.table_name, tc.constraint_name;

-- 6. Verify row counts
SELECT
    'documents' as table_name,
    (SELECT COUNT(*) FROM documents) as row_count
UNION ALL
SELECT
    'text_extraction_results',
    (SELECT COUNT(*) FROM text_extraction_results)
UNION ALL
SELECT
    'translation_results',
    (SELECT COUNT(*) FROM translation_results);
```
